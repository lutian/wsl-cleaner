cmake_minimum_required(VERSION 3.20)
project(WslCleaner VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(policy STATIC src/policy.cpp)
target_include_directories(policy PUBLIC include)
add_executable(core_checks tests/core_checks.cpp)
target_link_libraries(core_checks PRIVATE policy)
add_executable(maintenance_checks tests/maintenance_checks.cpp src/maintenance.cpp src/maintenance_policy.cpp)
target_link_libraries(maintenance_checks PRIVATE policy)
if(WIN32)
  add_compile_definitions(UNICODE _UNICODE)
  set(WINDOWS_LIBS user32 shell32 comctl32 ole32 advapi32 comdlg32 gdi32 bcrypt wintrust crypt32 uuid virtdisk)
  target_sources(core_checks PRIVATE src/core.cpp)
  target_sources(maintenance_checks PRIVATE src/core.cpp)
  target_link_libraries(core_checks PRIVATE ${WINDOWS_LIBS})
  target_link_libraries(maintenance_checks PRIVATE ${WINDOWS_LIBS})
  add_executable(virtual_disk_checks tests/virtual_disk_checks.cpp)
  target_link_libraries(virtual_disk_checks PRIVATE virtdisk ole32)
endif()
enable_testing()
add_test(NAME core COMMAND core_checks)
add_test(NAME maintenance COMMAND maintenance_checks)
if(WIN32)
  add_test(NAME isolated_virtual_disk COMMAND virtual_disk_checks)
endif()
